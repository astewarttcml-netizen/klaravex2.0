### `infra/rustdesk_controller/session.py`
```python
import subprocess
import os
import socket
from .protocol import RustDeskClient

class RustDeskSession(RustDeskClient):
    def __init__(self, peer_id: str, socket_path: str = '/tmp/rustdesk_socket'):
        self.peer_id = peer_id
        self.socket_path = socket_path
        self.handle = None

    def _connect(self, peer_id: str) -> int:
        # Start the RustDesk CLI subprocess
        self.process = subprocess.Popen(
            ['rustdesk-cli', '--connect', peer_id, '--socket', self.socket_path],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE
        )
        # Wait for the socket to be created
        while not os.path.exists(self.socket_path):
            time.sleep(0.1)
        # Connect to the socket
        self.socket = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        self.socket.connect(self.socket_path)
        # Return a handle (e.g., process PID)
        self.handle = self.process.pid
        return self.handle

    def _recv_frame(self, handle: int) -> bytes:
        # Receive a frame from the socket
        frame = self.socket.recv(4096)
        return frame

    def _send_event(self, handle: int, event: bytes) -> None:
        # Send an event to the socket
        self.socket.sendall(event)
```

import os
import time

def ensure_socket_path(socket_path: str) -> None:
    """
    Ensures the socket path exists and is removed if it already exists.
    
    :param socket_path: The path to the socket.
    """
    if os.path.exists(socket_path):
        os.remove(socket_path)
    time.sleep(0.1)  # Small delay to ensure the socket is removed
