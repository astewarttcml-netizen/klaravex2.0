## Entry Points

### `main.py`
```python
import json
from src.config import assistants
from importlib import import_module

def main():
    # Load configuration
    config = assistants.load_config()
    
    # Example request data
    intake_request = {
        "client_name": "Acme Corp",
        "project_name": "New Product Launch",
        "project_description": "Develop a new product for the consumer market.",
        "contact_info": {
            "email": "john.doe@acme.com",
            "phone": "123-456-7890"
        },
        "additional_notes": "Urgent project, needs to be completed by end of Q1."
    }
    
    engineering_task = {
        "task_name": "Design Prototype",
        "task_description": "Create a prototype for the new product.",
        "assigned_to": "Jane Smith",
        "due_date": "2023-12-31",
        "priority": 1,
        "additional_notes": "Ensure all safety standards are met."
    }
    
    # Handle intake request
    intake_module = import_module(config['assistants']['biz_intake']['module'])
    intake_function = getattr(intake_module, config['assistants']['biz_intake']['function'])
    intake_response = intake_function(intake_request)
    print("Intake Response:", json.dumps(intake_response, indent=2))
    
    # Handle engineering task
    engineering_module = import_module(config['assistants']['biz_engineer']['module'])
    engineering_function = getattr(engineering_module, config['assistants']['biz_engineer']['function'])
    engineering_response = engineering_function(engineering_task)
    print("Engineering Response:", json.dumps(engineering_response, indent=2))

if __name__ == "__main__":
    main()
```
