## Step 3 — Score Against Brand Criteria

> **Brand scoring dimensions (per HANDOFF A7):**
> | # | Dimension | Weight |
> |---|-----------|--------|
> | 1 | Color palette adherence (Klaravex palette) | 25 % |
> | 2 | Logo presence / placement | 20 % |
> | 3 | Typography / font compliance | 20 % |
> | 4 | Composition / safe-zone margins | 15 % |
> | 5 | Tone / mood alignment | 20 % |
>
> **Pass threshold: ≥ 70 / 100**

```bash
# POST each image to internal brand-score API
RESULTS_FILE="/tmp/t14_7_results.jsonl"
> "$RESULTS_FILE"

for img in /tmp/t14_7_images/*; do
  draft_id=$(basename "${img%.*}")
  echo "Scoring draft_id=${draft_id} ..."

  response=$(curl -fsSL --max-time 60 \
    -X POST "${BRAND_SCORE_ENDPOINT}/v1/score" \
    -H "Content-Type: multipart/form-data" \
    -F "file=@${img}" \
    -F "brand_profile=klaravex_v2" \
    -F "dimensions=color,logo,typography,composition,tone")

  echo "${response}" | jq \
    --arg id "$draft_id" \
    '. + {draft_id: $id}' \
    >> "$RESULTS_FILE"

  echo "  → score: $(echo $response | jq .total_score)"
done
```

## Step 4 — Parse Results & Flag Failures

```bash
PASS_THRESHOLD=70
FAIL_IDS=()

echo ""
echo "=== T14.7 Brand Score Summary ==="
printf "%-12s %-8s %-10s %s\n" "draft_id" "score" "status" "low_dimensions"

while IFS= read -r line; do
  draft_id=$(echo "$line" | jq -r '.draft_id')
  score=$(echo "$line"    | jq -r '.total_score')
  low_dims=$(echo "$line" | jq -r \
    '[.dimensions | to_entries[] | select(.value < 60) | .key] | join(",")')

  if (( $(echo "$score < $PASS_THRESHOLD" | bc -l) )); then
    status="FAIL ❌"
    FAIL_IDS+=("$draft_id")
  else
    status="PASS ✅"
  fi

  printf "%-12s %-8s %-10s %s\n" "$draft_id" "$score" "$status" "$low_dims"
done < "$RESULTS_FILE"

echo ""
echo "Failing draft IDs: ${FAIL_IDS[*]:-none}"
```

## Step 5 — Write Results to Assessment Table

```sql
-- Create table if not exists (run once)
CREATE TABLE IF NOT EXISTS klaravex_image_qa_assessments (
  id              SERIAL PRIMARY KEY,
  assessed_at     TIMESTAMPTZ DEFAULT NOW(),
  task_id         TEXT        NOT NULL,
  draft_id        TEXT        NOT NULL,
  total_score     NUMERIC(5,2),
  dim_color       NUMERIC(5,2),
  dim_logo        NUMERIC(5,2),
  dim_typography  NUMERIC(5,2),
  dim_composition NUMERIC(5,2),
  dim_tone        NUMERIC(5,2),
  pass_fail       TEXT,
  low_dimensions  TEXT,
  notes           TEXT
);
```

```bash

      (task_id, draft_id, total_score,

       pass_fail, low_dimensions)
    VALUES (

       $(echo $line | jq .total_score),
       $(echo $line | jq '.dimensions.color   // null'),
       $(echo $line | jq '.dimensions.logo    // null'),
       $(echo $line | jq '.dimensions.typography // null'),
       $(echo $line | jq '.dimensions.composition // null'),
       $(echo $line | jq '.dimensions.tone    // null'),
      '$(echo $line | jq -r "if .total_score >= $PASS_THRESHOLD then \"PASS\" else \"FAIL\" end")',
      '$(echo $line | jq -r "[.dimensions | to_entries[] | select(.value < 60) | .key] | join(\",\")")'
    );
  "
done < "$RESULTS_FILE"
```

## Step 6 — Flag Failing Drafts for Regeneration

```sql
-- Mark failing drafts in source table
UPDATE klaravex_social_drafts
SET
  draft_status = 'qa_failed',
  qa_notes     = 'T14.7: brand score below threshold — regeneration required',
  updated_at   = NOW()
WHERE id = ANY(
  SELECT draft_id::uuid
  FROM klaravex_image_qa_assessments
  WHERE task_id   = 'T14.7'
    AND pass_fail = 'FAIL'
);
```
